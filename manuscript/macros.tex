}

            \mathcal{Q}

            \right\}_{\mu=1}^{d_{\mathrm{QSE}}}, \\
            P_k
            &=
            \left\|
                \operatorname{Proj}_{\mathcal{Q}}
                \left|\Psi_k^{\mathrm{CASCI}}\right\rangle
            \right\|^2 .

        \label{eq:qse-exact-state-projection}
    \end{equation}%
}

% Second Quantised Electronic Hamiltonian
\newcommand{\ElectronicHamiltonianEquation}{%
    \begin{equation}
        \hat{H}
        =
        E_{\mathrm{nuc}}
        +
        \sum_{pq} h_{pq} \, \hat{a}^{\dagger}_p \hat{a}_q
        +
        \frac{1}{2}
        \sum_{pqrs} g_{pqrs} \,
        \hat{a}^{\dagger}_p \hat{a}^{\dagger}_q
        \hat{a}_s \hat{a}_r .
        \label{eq:electronic-hamiltonian}
    \end{equation}%
}
% Spin Z Operator
\newcommand{\SpinZEquation}{%
    \begin{equation}
        \hat{S}_z
        =
        \frac{1}{2}
        \sum_p
        \left(
            \hat{n}_{p\alpha}
            -
            \hat{n}_{p\beta}
        \right),
        \qquad
        \hat{n}_{p\sigma}
        =
        \hat{a}^{\dagger}_{p\sigma}
        \hat{a}_{p\sigma}.
        \label{eq:spin-z}
    \end{equation}%
}
% Spin Ladder Operators
\newcommand{\SpinLadderOperatorsEquation}{%
    \begin{equation}
        \hat{S}_+
        =
        \sum_p
        \hat{a}^{\dagger}_{p\alpha}
        \hat{a}_{p\beta},
        \quad
        \hat{S}_-
        =
        \sum_p
        \hat{a}^{\dagger}_{p\beta}
        \hat{a}_{p\alpha}.
        \label{eq:spin-ladder}
    \end{equation}%
}
% Total Spin Squared Operator
\newcommand{\SpinSquaredEquation}{%
    \begin{equation}
        \hat{S}^2
        =
        \hat{S}_z^2
        +
        \frac{1}{2}
        \left(
            \hat{S}_+\hat{S}_-
            +
            \hat{S}_-\hat{S}_+
        \right).
        \label{eq:spin-squared}
    \end{equation}%
}

\newcommand{\SuzukiTrotterFirstOrderEquation}{%
    \begin{equation}
        e^{\hat{T} - \hat{T}^{\dagger}}
        \approx
        \prod_{\lambda}
        e^{\theta_{\lambda}
        \left(\hat{T}_{\lambda} - \hat{T}_{\lambda}^{\dagger}\right)}.
        \label{eq:first-order-suzuki-trotter}
    \end{equation}%
}

% QSE OPERATORS
% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
% QSE Singlet Operator Pool
\newcommand{\QSESingletOperatorPoolEquation}{%
    \begin{align}
        \hat{O}^{S}_{ij}
        &=
        \hat{a}^{\dagger}_{i\alpha}\hat{a}_{j\alpha}
        +
        \hat{a}^{\dagger}_{i\beta}\hat{a}_{j\beta}, \\
        \mathcal{O}^{S}
        &=
        \left\{
            \hat{O}^{S}_{ij}
            \mathrel{\big|}
            i,j \in \{1, \ldots, n_{\mathrm{o}}\}
        \right\}.
        \label{eq:qse-singlet-operator-pool}
    \end{align}%
}
% QSE Triplet Operator Pool
\newcommand{\QSETripletOperatorPoolEquation}{%
    \begin{gather}
        \hat{O}^{T,0}_{ij}
        =
        \frac{1}{\sqrt{2}}
        \left(
            \hat{a}^{\dagger}_{i\alpha}\hat{a}_{j\alpha}
            -
            \hat{a}^{\dagger}_{i\beta}\hat{a}_{j\beta}
        \right),
        \qquad
        \hat{O}^{T,+1}_{ij}
        =
        \hat{a}^{\dagger}_{i\alpha}\hat{a}_{j\beta},
        \qquad
        \hat{O}^{T,-1}_{ij}
        =
        \hat{a}^{\dagger}_{i\beta}\hat{a}_{j\alpha}, \\
        \mathcal{O}^{T}
        =
        \left\{
            \hat{O}^{T,m}_{ij}
            \mathrel{\big|}
            i,j \in \{1, \ldots, n_{\mathrm{o}}\},
            \ m \in \{-1, 0, +1\}
        \right\}.
        \label{eq:qse-triplet-operator-pool}
    \end{gather}%
}

\newcommand{\QSEProjectedMatricesEquation}{%
    \begin{gather}
        H_{\mu\nu}
        =
        \left\langle\Psi_{\mathrm{ref}}\right|
        \hat{O}_\mu^\dagger \hat{H} \hat{O}_\nu
        \left|\Psi_{\mathrm{ref}}\right\rangle, \\
        S_{\mu\nu}
        =
        \left\langle\Psi_{\mathrm{ref}}\right|
        \hat{O}_\mu^\dagger \hat{O}_\nu
        \left|\Psi_{\mathrm{ref}}\right\rangle.
        \label{eq:qse-projected-matrices}
    \end{gather}%
}

% MATRIX OPERATIONS

% Diagonalisation of S
\newcommand{\DiagonaliseSEquation}{%
    \begin{equation}
        \mathbf{S}
        =
        \mathbf{U}
        \boldsymbol{\Lambda}
        \mathbf{U}^{\dagger}.
        \label{eq:diagonalise-s}
    \end{equation}%
}
% Whitening Equation
\newcommand{\WhitenSEquation}{%
    \begin{equation}
        \tilde{\mathbf{V}}
        =
        \tilde{\mathbf{U}}\tilde{\boldsymbol{\Lambda}}^{-1/2},
        \qquad
        \tilde{\mathbf{V}}^{\dagger}
        \mathbf{S}
        \tilde{\mathbf{V}}
        =
        \mathbf{I}.
        \label{eq:whiten-s}
    \end{equation}%
}

% Reduced Eigenvalue Equation
\newcommand{\ReducedEigenvalueEquation}{%
    \begin{equation}
        \tilde{\mathbf{V}}^{\dagger}
        \mathbf{H}
        \tilde{\mathbf{V}}
        \mathbf{y}
        =
        E\mathbf{y}.
        \label{eq:reduced-eigenvalue}
    \end{equation}%
}
% Max Dimensions (Stable SV)
\newcommand{\MaxDimensionsEquation}{%
    \begin{equation}
        \begin{aligned}
            d_{\mathrm{S}}^{\max}
            &=
            \min\!\left[
                \binom{n_{\mathrm{o}}}{n_{\mathrm{p}}}^{2}
                -
                \binom{n_{\mathrm{o}}}{n_{\mathrm{p}}+1}
                \binom{n_{\mathrm{o}}}{n_{\mathrm{p}}-1},
                \,
                n_{\mathrm{o}}^{2}
            \right], \\
            d_{\mathrm{T}}^{\max}
            &=
            \min\!\left[
                3\left(
                    \binom{n_{\mathrm{o}}}{n_{\mathrm{p}}+1}
                    \binom{n_{\mathrm{o}}}{n_{\mathrm{p}}-1}
                    -
                    \binom{n_{\mathrm{o}}}{n_{\mathrm{p}}+2}
                    \binom{n_{\mathrm{o}}}{n_{\mathrm{p}}-2}
                \right),
                \,
                3n_{\mathrm{o}}(n_{\mathrm{o}}-1)
            \right],
            \qquad
            n_{\mathrm{p}}=\frac{n_{\mathrm{e}}}{2}.
        \end{aligned}
        \label{eq:max-dimensions}
    \end{equation}%
}

% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
% SPIN Expectation
% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
% Spin Expectation in QSE Basis
\newcommand{\SpinExpectationQSEBasisEquation}{%
    \begin{equation}
        \left\langle \hat{S}^2 \right\rangle_k
        =
        \frac{\mathbf{c}_k^\dagger\mathbf{Z}\mathbf{c}_k}
             {\mathbf{c}_k^\dagger\mathbf{S}\mathbf{c}_k}.
        \label{eq:spin-expectation-qse-basis}
    \end{equation}%
}
% Spin Expectation in Reduced Basis
\newcommand{\SpinExpectationShortenedBasisEquation}{%
    \begin{equation}
        \left\langle \hat{S}^2 \right\rangle_k
        =
        \frac{
            \mathbf{y}_k^\dagger
            \tilde{\mathbf{V}}^\dagger\mathbf{Z}\tilde{\mathbf{V}}
            \mathbf{y}_k
        }{
            \mathbf{y}_k^\dagger\mathbf{y}_k
        }.
        \label{eq:spin-expectation-shortened-basis}
    \end{equation}%
}

% Singlet operator acting on Hartree--Fock
\newcommand{\SingletOnHFEquation}{%
    \begin{equation}
        \hat{O}^{S}_{ii}
        \left|\mathrm{HF}\right\rangle
        =
        \left(
            \hat{n}_{i\alpha}
            +
            \hat{n}_{i\beta}
        \right)
        \left|\mathrm{HF}\right\rangle
        =
        2\left|\mathrm{HF}\right\rangle .
        \label{eq:singlet-on-hf}
    \end{equation}%
}

% Triplet operator acting on Hartree--Fock
\newcommand{\TripletOnHFEquation}{%
    \begin{equation}
        \hat{O}^{T,0}_{ii}
        \left|\mathrm{HF}\right\rangle
        =
        \left(
            \hat{n}_{i\alpha}
            -
            \hat{n}_{i\beta}
        \right)
        \left|\mathrm{HF}\right\rangle
        =
        0 .
        \label{eq:triplet-on-hf}
    \end{equation}%
}
